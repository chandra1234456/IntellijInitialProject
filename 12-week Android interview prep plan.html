P1 · Implement a generic SafeResult wrapper (like Arrow's Either) Medium
Implement a sealed class Result<T> with Success and Error subtypes. Add map(), flatMap(), getOrElse(), and onSuccess/onError extension functions. This pattern is used in every production Android repository layer. 
sealed class Result<out T> {
    data class Success<T>(val data: T) : Result<T>()
    data class Error(val exception: Throwable, val msg: String = exception.message.orEmpty()) : Result<Nothing>()
}

// Extension functions
inline fun <T, R> Result<T>.map(transform: (T) -> R): Result<R> = when (this) {
    is Result.Success -> Result.Success(transform(data))
    is Result.Error   -> this
}

inline fun <T, R> Result<T>.flatMap(transform: (T) -> Result<R>): Result<R> = when (this) {
    is Result.Success -> transform(data)
    is Result.Error   -> this
}

fun <T> Result<T>.getOrElse(default: T): T = when (this) {
    is Result.Success -> data
    is Result.Error   -> default
}

inline fun <T> Result<T>.onSuccess(action: (T) -> Unit): Result<T> {
    if (this is Result.Success) action(data)
    return this
}

// Usage in Repository
suspend fun getUser(id: String): Result<User> = try {
    Result.Success(api.fetchUser(id))
} catch (e: Exception) {
    Result.Error(e)
}
P2 · Implement LRU Cache in Kotlin from scratch Hard
Implement a thread-safe LRU Cache with get(key) and put(key, value) in O(1) time. Use LinkedHashMap's access-order feature. Common in memory-management discussions. 
class LRUCache<K, V>(private val capacity: Int) {

    // accessOrder=true: get() moves entry to tail (most recently used)
    private val cache = object : LinkedHashMap<K, V>(16, 0.75f, true) {
        override fun removeEldestEntry(eldest: Map.Entry<K, V>) = size > capacity
    }

    @Synchronized
    fun get(key: K): V? = cache[key]

    @Synchronized
    fun put(key: K, value: V) { cache[key] = value }

    @Synchronized
    fun remove(key: K) = cache.remove(key)

    val size: Int get() = cache.size
}

// For Coroutines-friendly version, use Mutex instead of @Synchronized:
class CoroutineLRUCache<K, V>(private val capacity: Int) {
    private val mutex = Mutex()
    private val cache = LinkedHashMap<K, V>(16, 0.75f, true)

    suspend fun get(key: K): V? = mutex.withLock { cache[key] }

    suspend fun put(key: K, value: V) = mutex.withLock {
        cache[key] = value
        if (cache.size > capacity) cache.remove(cache.keys.first())
    }
}
P3 · Implement a ViewModel that exposes StateFlow from LiveData Easy
Write a ViewModel that fetches a list of User objects from a repository, exposes a UiState sealed class via StateFlow, and handles loading/error/success states. 
sealed class UiState<out T> {
    object Loading : UiState<Nothing>()
    data class Success<T>(val data: T) : UiState<T>()
    data class Error(val message: String) : UiState<Nothing>()
}

class UsersViewModel(private val repo: UserRepository) : ViewModel() {

    private val _uiState = MutableStateFlow<UiState<List<User>>>(UiState.Loading)
    val uiState = _uiState.asStateFlow()

    // Single-shot events (navigation, snackbar) via Channel
    private val _events = Channel<UserEvent>(Channel.BUFFERED)
    val events = _events.receiveAsFlow()

    init { loadUsers() }

    private fun loadUsers() {
        viewModelScope.launch {
            _uiState.value = UiState.Loading
            _uiState.value = try {
                UiState.Success(repo.getUsers())
            } catch (e: Exception) {
                UiState.Error(e.message ?: "Unknown error")
            }
        }
    }

    fun onUserClicked(user: User) {
        viewModelScope.launch {
            _events.send(UserEvent.Navigate(user.id))
        }
    }
}

// Collect in Fragment:
viewLifecycleOwner.lifecycleScope.launch {
    repeatOnLifecycle(Lifecycle.State.STARTED) {
        viewModel.uiState.collect { state ->
            when (state) {
                is UiState.Loading -> showLoading()
                is UiState.Success -> showUsers(state.data)
                is UiState.Error   -> showError(state.message)
            }
        }
    }
}

P4 · Implement a search-as-you-type with Flow + debounce Medium
Build a ViewModel where a search query StateFlow is debounced by 300ms, then passed to a repository. Use flatMapLatest to cancel in-flight searches on new input. This is one of the most asked real-world Flow questions. 
class SearchViewModel(private val repo: SearchRepository) : ViewModel() {

    private val _query = MutableStateFlow("")

    val results: StateFlow<UiState<List<SearchResult>>> = _query
        .debounce(300L)              // wait 300ms after last keystroke
        .filter { it.isNotBlank() }  // ignore empty queries
        .distinctUntilChanged()       // skip identical consecutive queries
        .flatMapLatest { query ->    // cancel previous search on new query
            flow {
                emit(UiState.Loading)
                emit(try {
                    UiState.Success(repo.search(query))
                } catch (e: Exception) {
                    UiState.Error(e.message.orEmpty())
                })
            }
        }
        .stateIn(
            scope = viewModelScope,
            started = SharingStarted.WhileSubscribed(5_000),
            initialValue = UiState.Success(emptyList())
        )

    fun onQueryChanged(q: String) { _query.value = q }
}
P5 · Implement retry with exponential backoff using Flow Hard
Write an extension function Flow<T>.retryWithBackoff(maxRetries, initialDelay) that retries on failure with exponentially increasing delays. Used in network layers. 
fun <T> Flow<T>.retryWithBackoff(
    maxRetries: Int = 3,
    initialDelayMs: Long = 1_000L,
    maxDelayMs: Long = 30_000L,
    factor: Double = 2.0,
    shouldRetry: (Throwable) -> Boolean = { true }
): Flow<T> = this.retryWhen { cause, attempt ->
    if (attempt < maxRetries && shouldRetry(cause)) {
        val delayMs = minOf(
            initialDelayMs * factor.pow(attempt.toDouble()).toLong(),
            maxDelayMs
        )
        delay(delayMs)
        true  // retry
    } else {
        false  // give up
    }
}

// Usage:
flow { emit(api.fetchData()) }
    .retryWithBackoff(
        maxRetries = 3,
        shouldRetry = { it is IOException }  // only retry network errors
    )
    .catch { e -> emit(fallbackData()) }
    .collect { render(it) }
P6 · Implement a thread-safe single-instance pattern using coroutines Hard
Implement a class that lazily initializes a resource (e.g. DB connection) exactly once, even when called from multiple coroutines concurrently. No double-initialization allowed. 
class SingletonResource {
    private val mutex = Mutex()
    private var resource: ExpensiveResource? = null

    suspend fun getResource(): ExpensiveResource {
        resource?.let { return it }  // fast path — no lock needed
        return mutex.withLock {
            // double-checked locking inside the mutex
            resource ?: createResource().also { resource = it }
        }
    }

    private suspend fun createResource(): ExpensiveResource {
        delay(100)  // simulating async init (DB open, auth, etc.)
        return ExpensiveResource()
    }
}

// Alternative: use Kotlin's built-in lazy with coroutines:
class LazyResource(private val scope: CoroutineScope) {
    private val deferred: Deferred<ExpensiveResource> = scope.async(start = CoroutineStart.LAZY) {
        createResource()
    }
    suspend fun get() = deferred.await()  // safe: only initializes once
}

P7 · Implement a BFS-based shortest path on a social graph Medium
Given a social network as an adjacency list Map<String, List<String>>, find the shortest connection path between two users. Return the path as a list of names. This tests both DSA and Kotlin collections fluency. 
fun shortestPath(
    graph: Map<String, List<String>>,
    start: String,
    end: String
): List<String>? {
    if (start == end) return listOf(start)

    val queue   = ArrayDeque<List<String>>()  // stores paths, not nodes
    val visited = mutableSetOf(start)

    queue.addLast(listOf(start))

    while (queue.isNotEmpty()) {
        val path    = queue.removeFirst()
        val current = path.last()

        for (neighbor in graph[current].orEmpty()) {
            if (neighbor == end) return path + neighbor
            if (neighbor !in visited) {
                visited.add(neighbor)
                queue.addLast(path + neighbor)
            }
        }
    }
    return null  // no path found
}

// Test:
val graph = mapOf(
    "Alice" to listOf("Bob", "Carol"),
    "Bob"   to listOf("Dave"),
    "Carol" to listOf("Dave", "Eve"),
    "Dave"  to listOf("Frank"),
    "Eve"   to listOf("Frank")
)
shortestPath(graph, "Alice", "Frank")
// [Alice, Carol, Eve, Frank] or [Alice, Carol, Dave, Frank]
P8 · Design & implement a simple image loading cache Hard
Implement a simple 2-level cache (memory + disk) image loader in Kotlin. Use coroutines for async loading, an LRU memory cache, and a disk cache with a file naming strategy. This is the Glide/Coil system design question reduced to code. 
class ImageLoader(
    private val cacheDir: File,
    memoryCacheSize: Int = 50
) {
    // L1: Memory cache — LRU
    private val memCache = LRUCache<String, Bitmap>(memoryCacheSize)

    // L2: Disk cache key = MD5(url)
    private fun diskKey(url: String) = url.md5()

    suspend fun load(url: String): Bitmap? = withContext(Dispatchers.IO) {
        // 1. Check memory cache
        memCache.get(url)?.let { return@withContext it }

        // 2. Check disk cache
        val diskFile = File(cacheDir, diskKey(url))
        if (diskFile.exists()) {
            val bitmap = BitmapFactory.decodeFile(diskFile.path)
            bitmap?.let { memCache.put(url, it) }
            return@withContext bitmap
        }

        // 3. Network fetch
        return@withContext try {
            val bytes = fetchBytes(url)
            val bitmap = BitmapFactory.decodeByteArray(bytes, 0, bytes.size)
            bitmap?.let {
                diskFile.writeBytes(bytes)  // save to disk
                memCache.put(url, it)        // save to memory
            }
            bitmap
        } catch (e: Exception) { null }
    }

    private suspend fun fetchBytes(url: String): ByteArray {
        return URL(url).openStream().use { it.readBytes() }
    }
}

fun String.md5(): String {
    val md = MessageDigest.getInstance("MD5")
    return md.digest(toByteArray()).joinToString("") { "%02x".format(it) }
}

P10 · Implement a token refresh Authenticator for OkHttp Hard
Implement an OkHttp Authenticator that refreshes an expired JWT token, queues concurrent requests during refresh, and retries them with the new token. This is asked in every senior Android role. 
class TokenAuthenticator(
    private val tokenStore: TokenStore,
    private val authApi: AuthApi
) : Authenticator {

    // Mutex ensures only ONE refresh happens even with concurrent 401s
    private val mutex = Mutex()

    override fun authenticate(route: Route?, response: Response): Request? {
        // If token already changed since this request was made → retry with new token
        val savedToken = tokenStore.getAccessToken()
        val requestToken = response.request.header("Authorization")
            ?.removePrefix("Bearer ")

        if (savedToken != null && savedToken != requestToken) {
            return response.request.newBuilder()
                .header("Authorization", "Bearer $savedToken")
                .build()
        }

        // Refresh token — block thread (OkHttp is on IO thread)
        val newToken = runBlocking {
            mutex.withLock {
                // Check again inside lock — another coroutine may have refreshed
                val current = tokenStore.getAccessToken()
                if (current != requestToken) return@withLock current

                try {
                    val result = authApi.refreshToken(tokenStore.getRefreshToken()!!)
                    tokenStore.saveTokens(result.accessToken, result.refreshToken)
                    result.accessToken
                } catch (e: Exception) {
                    tokenStore.clearTokens()
                    null  // triggers logout
                }
            }
        } ?: return null

        return response.request.newBuilder()
            .header("Authorization", "Bearer $newToken")
            .build()
    }
}
